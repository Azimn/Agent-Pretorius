def _calculate_salience(self, directive: Dict, state: Dict) -> float:
 """
 Calculate directive salience (urgency score).
 
 Returns: base_priority + weighted state/belief contributions
 """
 score = directive.get('base_priority', directive.get('priority', 0))
 
 for modifier in directive.get('salience_modifiers', []):
 if 'variable' in modifier:
 var_value = state['variables'].get(modifier['variable'], 0)
 score += var_value * modifier['weight']
 
 elif 'belief' in modifier:
 belief_value = state['beliefs'].get(modifier['belief'], 0)
 score += belief_value * modifier['weight']
 
 return score
def _find_directive(self, user_input: str, state: Dict, events: List[str]) -> Dict:
 """Find highest-salience matching directive."""
